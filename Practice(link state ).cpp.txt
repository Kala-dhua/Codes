#include <iostream>
using namespace std;

#define N 4
#define INF 999

int main(){
	int cost[N][N]={
		{0, 3, INF, 7},
		{3,   0,   1,  INF},
        {INF, 1,   0,   2},
        {7,   INF, 2,   0}
	
	};
	
	int dist[N][N];
	for(int i=0; i<N; i++){
		for(int j=0; j<N; j++){
			dist[i][j]=cost[i][j];
		}
	}
	
	for(int k=0; k<N; k++){
		for(int i=0; i<N; i++){
			for(int j=0; j<N; j++){
				if(cost[i][k] !=INF && dist[k][j]!=INF){
					if(dist[i][k]+dist[k][j] < dist[i][j]){
						dist[i][j]= dist[i][k]+dist[k][j];
					}
				}
			}
		}
	}
	
	for(int i=0; i<N; i++){
		cout<<"Routing table for rouer "<<i<<endl;
		cout<<"Destination\tCost"<<endl;
		for(int j=0; j<N; j++){
			cout<<j<< "\t\t" <<dist[i][j];
		}
	}
