#include <iostream>
#include <cmath>
using namespace std;

// Function to calculate number of parity bits needed
int calculateParityBits(int m)
{
    int r = 0;

    while (pow(2, r) < (m + r + 1))
    {
        r++;
    }

    return r;
}

// Function to insert data bits into positions
void insertDataBits(int data[], int hamming[], int m, int r)
{
    int j = 0;

    for (int i = 1; i <= m + r; i++)
    {
        // Power of 2 positions are parity bit positions
        if ((i & (i - 1)) == 0)
        {
            hamming[i] = -1;
        }
        else
        {
            hamming[i] = data[j++];
        }
    }
}

// Function to calculate parity bits
void calculateParity(int hamming[], int r, int total)
{
    for (int i = 0; i < r; i++)
    {
        int parityPos = pow(2, i);
        int count = 0;

        for (int j = 1; j <= total; j++)
        {
            if (j & parityPos)
            {
                if (hamming[j] != -1)
                {
                    count += hamming[j];
                }
            }
        }

        // Even parity
        hamming[parityPos] = (count % 2 == 0) ? 0 : 1;
    }
}

// Function to detect and correct single-bit error
int detectError(int hamming[], int r, int total)
{
    int errorPos = 0;

    for (int i = 0; i < r; i++)
    {
        int parityPos = pow(2, i);
        int count = 0;

        for (int j = 1; j <= total; j++)
        {
            if (j & parityPos)
            {
                count += hamming[j];
            }
        }

        if (count % 2 != 0)
        {
            errorPos += parityPos;
        }
    }

    return errorPos;
}

int main()
{
    int m;

    cout << "Enter number of data bits: ";
    cin >> m;

    int data[m];

    cout << "Enter data bits (0/1) one by one:" << endl;

    for (int i = 0; i < m; i++)
    {
        cin >> data[i];
    }

    int r = calculateParityBits(m);
    int total = m + r;

    int hamming[total + 1];

    // Insert data bits
    insertDataBits(data, hamming, m, r);

    // Calculate parity bits
    calculateParity(hamming, r, total);

    cout << "\nGenerated Hamming Code: ";

    for (int i = 1; i <= total; i++)
    {
        cout << hamming[i];
    }

    cout << endl;

    // Introduce an error
    int errPos;

    cout << "\nEnter position to introduce error (1 to "
         << total << "), or 0 for no error: ";

    cin >> errPos;

    if (errPos >= 1 && errPos <= total)
    {
        // Flip the bit
        hamming[errPos] = hamming[errPos] ^ 1;

        cout << "Bit at position " << errPos
             << " flipped. Transmitted (corrupted) code: ";

        for (int i = 1; i <= total; i++)
        {
            cout << hamming[i];
        }

        cout << endl;
    }

    // Detect error
    int error = detectError(hamming, r, total);

    if (error == 0)
    {
        cout << "\nNo error detected." << endl;
    }
    else
    {
        cout << "\nError detected at position: "
             << error << endl;

        // Correct the error
        hamming[error] = hamming[error] ^ 1;

        cout << "Corrected Hamming Code: ";

        for (int i = 1; i <= total; i++)
        {
            cout << hamming[i];
        }

        cout << endl;
    }

    return 0;
}
