#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;
#define frames 10

int main(){
	int frame = 0;
	int seq_no = 0;
	srand(time(NULL));
	
	cout<<"--------------Stop and Wait Simulator----------"<<endl;
	
	while(frame < frames){
		cout<<"[Sender] sending frame "<<frame<<" , "<<"sequence no : "<<seq_no;
		int network_event = rand() % 10;
		if(network_event < 3){
			cout<<"\n\n[Network] : Frame lost/Timeout expired..."<<endl;
			cout<<"sender retransmitting frame "<<frame<<endl;
		}
		else{
			cout<<"\n\n[Receiver] received frame succesfully and sending ack "<<1-seq_no<<"...."<<endl;
			cout<<"[sender] received ack "<<1-seq_no<<" successfully!"<<endl;
			frame++;
			seq_no = 1-seq_no;
		}
		
	}
	cout<<"All frames sent and acknowledge successfully!"<<endl;
	return 0;
}
